# Offline tool, not part of the game build. Pins recastnavigation v1.6.0 (MIT).
cmake_minimum_required(VERSION 3.16)
project(recast_bake CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
set(RECAST_SRC "" CACHE PATH "local recastnavigation checkout (optional; otherwise fetched)")
if(RECAST_SRC)
  set(RN ${RECAST_SRC})
else()
  FetchContent_Declare(rn GIT_REPOSITORY https://github.com/recastnavigation/recastnavigation GIT_TAG v1.6.0)
  FetchContent_GetProperties(rn)
  if(NOT rn_POPULATED)
    FetchContent_Populate(rn)
  endif()
  set(RN ${rn_SOURCE_DIR})
endif()
file(GLOB RC_SRC ${RN}/Recast/Source/*.cpp ${RN}/Detour/Source/*.cpp)
add_executable(recast_bake recast_bake.cpp ${RC_SRC})
target_include_directories(recast_bake PRIVATE ${RN}/Recast/Include ${RN}/Detour/Include)
find_package(ZLIB REQUIRED)
target_link_libraries(recast_bake PRIVATE ZLIB::ZLIB)

include(CTest)
if(BUILD_TESTING)
  add_executable(retain_box_test retain_box_test.cpp ${RC_SRC})
  target_include_directories(retain_box_test PRIVATE ${RN}/Recast/Include ${RN}/Detour/Include)
  target_link_libraries(retain_box_test PRIVATE ZLIB::ZLIB)
  add_test(NAME retain_box_loader COMMAND retain_box_test ${CMAKE_CURRENT_SOURCE_DIR}/water_fixtures/retain-box.obj)
endif()
